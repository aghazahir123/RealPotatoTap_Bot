// models/User.js - User & Admin Roles const mongoose = require('mongoose'); const userSchema = new mongoose.Schema({ telegramId: { type: String, required: true, unique: true }, username: String, role: { type: String, enum: ['user', 'team_member', 'admin'], default: 'user' }, permissions: { canManageTasks: { type: Boolean, default: false }, canEditUsers: { type: Boolean, default: false }, canSetCombos: { type: Boolean, default: false }, }, coins: { type: Number, default: 0 }, profitPerHour: { type: Number, default: 0 }, energy: { type: Number, default: 1000 }, maxEnergy: { type: Number, default: 1000 }, lastHarvest: { type: Date, default: Date.now }, walletAddress: { type: String, default: "" }, referrals: [{ type: mongoose.Schema.Types.ObjectId, ref: 'User' }] }); module.exports = mongoose.model('User', userSchema);
// middleware/auth.js const User = require('../models/User'); // Permission Middleware Check const checkPermission = (requiredPermission) => { return async (req, res, next) => { const adminUser = await User.findOne({ telegramId: req.headers['admin-id'] }); if (!adminUser) return res.status(403).json({ error: "Access denied." }); // Super Admin (Aap) ko har cheez ki full access hai if (adminUser.role === 'admin') return next(); // Team Member ke paas specific permission honi chahiye if (adminUser.role === 'team_member' && adminUser.permissions[requiredPermission]) { return next(); } return res.status(403).json({ error: "Permission Denied: Aapke paas is kaam ka haq nahi hai!" }); }; }; module.exports = checkPermission; // config/admin.js // AAP KI PERMANENT SUPER ADMIN ID const SUPER_ADMIN_ID = "7208905905"; const isSuperAdmin = (telegramId) => { return String(telegramId) === SUPER_ADMIN_ID; }; const checkAdminRights = (requiredPermission = null) => { return async (req, res, next) => { const userId = req.headers['x-telegram-user-id']; if (!userId) { return res.status(401).json({ success: false, message: "Unauthorized Request" }); } // 1. Agar AAP (7208905905) hain, toh DIRECT ACCESS! if (isSuperAdmin(userId)) { req.isAdmin = true; req.isSuperAdmin = true; return next(); } // 2. Agar Team Member hai, toh Unki Allowed Permissions Check Hogi const user = await User.findOne({ telegramId: userId }); if (user && user.role === 'team_member') { if (requiredPermission && user.permissions[requiredPermission] === true) { req.isAdmin = true; req.isSuperAdmin = false; return next(); } else { return res.status(403).json({ success: false, message: "Aapke paas is feature ko use karne ka haq nahi hai!" }); } } return res.status(403).json({ success: false, message: "Access Denied!" }); }; }; module.exports = { SUPER_ADMIN_ID, isSuperAdmin, checkAdminRights };
